cmake_minimum_required(VERSION 3.16)
project(shuangsheng VERSION 0.2.0 LANGUAGES CXX)
find_package(Fcitx5Core REQUIRED)
include(GNUInstallDirs)
get_filename_component(DEFAULT_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SHUANGSHENG_PROJECT_ROOT "${DEFAULT_PROJECT_ROOT}" CACHE PATH "Installed checkout containing ime and .venv")
configure_file(paths.h.in paths.h @ONLY)
add_library(shuangsheng MODULE shuangsheng.cpp)
target_compile_features(shuangsheng PRIVATE cxx_std_17)
target_compile_options(shuangsheng PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(shuangsheng PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(shuangsheng PRIVATE Fcitx5::Core)
set_target_properties(shuangsheng PROPERTIES PREFIX "")
install(TARGETS shuangsheng DESTINATION "${CMAKE_INSTALL_LIBDIR}/fcitx5")
install(FILES shuangsheng-addon.conf DESTINATION "${CMAKE_INSTALL_DATADIR}/fcitx5/addon" RENAME shuangsheng.conf)
install(FILES shuangsheng.conf DESTINATION "${CMAKE_INSTALL_DATADIR}/fcitx5/inputmethod")
